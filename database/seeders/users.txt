<?php

namespace Database\Seeders;

use Illuminate\Database\Seeder;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\File;
use Illuminate\Support\Facades\Hash;

class UsersTableSeeder extends Seeder
{
    public function run(): void
    {
        $sqlPath = base_path('users.sql');

        if (! File::exists($sqlPath)) {
            $this->command?->warn('users.sql not found: '.$sqlPath);

            return;
        }

        $sql = File::get($sqlPath);
        preg_match_all(
            '~INSERT INTO "public"\."users"\s*\(([^)]+)\)\s*VALUES\s*\((.+?)\);~s',
            $sql,
            $matches,
            PREG_SET_ORDER
        );

        if ($matches === []) {
            $this->command?->warn('No user inserts found in users.sql');

            return;
        }

        $seeded = 0;

        foreach ($matches as $match) {
            $columns = $this->parseColumns($match[1]);
            $values = $this->parseValuesRow($match[2]);

            if (count($columns) !== count($values)) {
                $this->command?->warn('Column/value count mismatch for one users.sql row.');
                continue;
            }

            $data = array_combine($columns, $values);

            unset($data['balance'], $data['id']);
            if (empty($data['password'])) {
                $data['password'] = Hash::make('azm999');
            }

            if (empty($data['user_name'])) {
                continue;
            }

            DB::table('users')->updateOrInsert(
                ['user_name' => $data['user_name']],
                $data
            );

            $seeded++;
        }

        $this->command?->info("Seeded {$seeded} users from users.sql");
    }

    private function parseColumns(string $columnsRaw): array
    {
        return array_map(
            fn ($column) => trim($column, " \n\r\t\""),
            explode(',', $columnsRaw)
        );
    }

    private function parseValuesRow(string $row): array
    {
        $values = [];
        $current = '';
        $inString = false;
        $length = strlen($row);

        for ($i = 0; $i < $length; $i++) {
            $char = $row[$i];

            if ($inString) {
                if ($char === "'") {
                    if ($i + 1 < $length && $row[$i + 1] === "'") {
                        $current .= "'";
                        $i++;
                        continue;
                    }
                    $inString = false;
                    continue;
                }

                $current .= $char;
                continue;
            }

            if ($char === "'") {
                $inString = true;
                continue;
            }

            if ($char === ',') {
                $values[] = $this->normalizeSqlValue($current);
                $current = '';
                continue;
            }

            $current .= $char;
        }

        $values[] = $this->normalizeSqlValue($current);

        return $values;
    }

    private function normalizeSqlValue(string $value): mixed
    {
        $value = trim($value);

        if (strtoupper($value) === 'NULL') {
            return null;
        }

        if ($value === 't') {
            return true;
        }

        if ($value === 'f') {
            return false;
        }

        if (is_numeric($value)) {
            return str_contains($value, '.') ? (float) $value : (int) $value;
        }

        return $value;
    }
}
